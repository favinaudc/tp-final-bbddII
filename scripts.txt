# V6: Cambio de esquema en línea

    ## Preparación (agregar más registros de onda):

        INSERT INTO biocustodia.contenedores_crio (
            lote_id, codigo_contenedor, laboratorio_origen, compuesto,
            temperatura_c, dosis_disponibles, estado_custodia, ultimo_control
        )
        SELECT
            g,
            'CRIO-LOT-' || lpad(g::STRING, 6, '0'),
            CASE (g % 5) WHEN 0 THEN 'BioSur Patagonia' WHEN 1 THEN 'Instituto Austral' WHEN 2 THEN 'CryoLab Córdoba' WHEN 3 THEN 'Genoma Norte' ELSE 'Biotech Cuyo' END,
            CASE (g % 4) WHEN 0 THEN 'Vacuna ARNm-V4' WHEN 1 THEN 'Plasma Hiperinmune' WHEN 2 THEN 'Células Madre CD34+' ELSE 'Enzima Polimerasa Taq' END,
            (-20.00 - (g % 175))::DECIMAL(5,2),
            10 + (g % 990),
            CASE (g % 4) WHEN 0 THEN 'ALMACENADO' WHEN 1 THEN 'EN_TRANSITO' WHEN 2 THEN 'CUARENTENA' ELSE 'DESPACHADO' END,
            now() - ((g % 720) || ' hours')::INTERVAL
        FROM generate_series(100006, 250005) AS g
        ON CONFLICT (lote_id) DO NOTHING;

    ## Pestaña 1 (creación de indices):
